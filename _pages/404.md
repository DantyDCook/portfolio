<script>
// ===== CONFIG =====
const DEFAULT_REDIRECT_DELAY = 3000; // ms (0 = instant)
const LOADER = "dots";               // "dots" | "spinner"
// ==================

// Normalize path (strip trailing slash except root)
const rawPath = location.pathname;
const p = rawPath.length > 1 ? rawPath.replace(/\/+$/, "") : "/";

// Vanity redirects map (add more as needed)
const VANITY = new Map([
  ["/linkedin", "https://www.linkedin.com/in/dantycook"],
  ["/github",   "https://github.com/dantydcook"],
  ["/portfolio", "https://www.dantydcook.com"]
  ["/resume",   "https://www.dantydcook.com/resume"],
  ["/hireme",   "mailto:hireme@dantydcook.com"],
]);

// Helper: does the path equal a key or key + trailing slash?
function vanityLookup(pathname) {
  if (VANITY.has(pathname)) return VANITY.get(pathname);
  // also match with trailing slash
  const noSlash = pathname.endsWith("/") ? pathname.slice(0, -1) : pathname;
  const withSlash = pathname.endsWith("/") ? pathname : pathname + "/";
  if (VANITY.has(noSlash)) return VANITY.get(noSlash);
  if (VANITY.has(withSlash)) return VANITY.get(withSlash);
  return null;
}

// Determine target + delay
let target = "/";
let delay = DEFAULT_REDIRECT_DELAY;

// 1) Vanity paths (instant redirect)
const vanityTarget = vanityLookup(p);
if (vanityTarget) {
  target = vanityTarget;
  delay = 0; // make vanity links snappy
}

// 2) Fix legacy /portfolio/* links → strip "/portfolio" prefix
else if (p.startsWith("/portfolio/")) {
  target = p.replace("/portfolio", "") + location.search + location.hash;
  // keep configured delay
}

// 3) (Optional) Recover from accidentally-prefixed external URLs
// e.g. /linkedin.com/in/dantycook → https://linkedin.com/in/dantycook
else if (/^\/(www\.)?(linkedin\.com|github\.com)(\/|$)/i.test(p)) {
  target = "https://" + p.replace(/^\//, "");
  delay = 0;
}

// 4) Fallback to home
else {
  target = "/";
}

// Redirect function (mailto should use href, not replace)
let redirectTimeout;
let seconds = Math.ceil(Math.max(0, delay) / 1000);

function doRedirect() {
  if (redirectTimeout) clearTimeout(redirectTimeout);
  if (target.startsWith("mailto:")) {
    // mailto works best with href navigation
    window.location.href = target;
  } else {
    // use replace to avoid 404 in history
    window.location.replace(target);
  }
}

document.addEventListener("DOMContentLoaded", () => {
  const msg = document.getElementById("redirectMsg");
  const live = document.getElementById("live");
  const manualLink = document.getElementById("manualLink");
  const spinner = document.querySelector(".spinner");
  const dots = document.querySelector(".dots");
  const btn = document.getElementById("homeBtn");

  manualLink.href = target;
  manualLink.textContent = target.startsWith("mailto:") ? "email me" : "click here";
  btn.addEventListener("click", doRedirect);

  // Toggle loader
  if (LOADER === "spinner") spinner.style.display = "block";
  else dots.style.display = "flex";

  if (delay > 0) {
    msg.textContent = `You’ll be redirected in ${seconds} second${seconds > 1 ? "s" : ""}…`;
    live.textContent = msg.textContent;

    const interval = setInterval(() => {
      seconds--;
      if (seconds > 0) {
        const text = `You’ll be redirected in ${seconds} second${seconds > 1 ? "s" : ""}…`;
        msg.textContent = text;
        live.textContent = text;
      } else {
        clearInterval(interval);
      }
    }, 1000);

    redirectTimeout = setTimeout(doRedirect, delay);
  } else {
    msg.textContent = "Redirecting…";
    live.textContent = "Redirecting now.";
    doRedirect();
  }
});
</script>
